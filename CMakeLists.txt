cmake_minimum_required(VERSION 3.24)
project(game_vm LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VM_FUZZ "Build the Clang libFuzzer harness" OFF)
add_library(vm src/vm.cpp)
target_include_directories(vm PUBLIC include)
add_executable(game_vm cli.cpp)
target_link_libraries(game_vm PRIVATE vm)
if(BUILD_TESTING)
    include(FetchContent)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.16.0)
    FetchContent_MakeAvailable(googletest)
    add_executable(vm_tests tests.cpp)
    target_link_libraries(vm_tests PRIVATE vm GTest::gtest_main)
    add_test(NAME vm_tests COMMAND vm_tests)
    add_test(NAME gameplay COMMAND game_vm run "${CMAKE_CURRENT_SOURCE_DIR}/examples/wave.gs")
    set_tests_properties(gameplay PROPERTIES PASS_REGULAR_EXPRESSION "result: 138")
endif()
if(VM_FUZZ)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR MSVC)
        message(FATAL_ERROR "Build the fuzz harness with clang++ on Linux")
    endif()
    add_executable(vm_fuzz fuzz.cpp src/vm.cpp)
    target_include_directories(vm_fuzz PRIVATE include)
    target_compile_options(vm_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(vm_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
